  <!-- 图表容器 -->
  <svg id="chart" width="800" height="500"></svg>

  <script>
    // 这里改数据文件名：iris.csv 或 dateforD3.csv
    const csvPath = "iris.csv";

    // 这里改用于降维/归一化的两列
    const xKey = "Sepal.Length"; // X轴字段
    const yKey = "Petal.Length"; // Y轴字段

    // 这里改点的大小和颜色
    const dotRadius = 4;
    const dotColor = "#4daf4a";

    // 这里改图表留白和画布大小
    const svg = d3.select("#chart");
    const margin = 30;
    const width = +svg.attr("width") - margin * 2;
    const height = +svg.attr("height") - margin * 2;
    const g = svg.append("g").attr("transform", `translate(${margin},${margin})`);

    // 归一化：把原始数据压到 0~1
    const normalize = (value, extent) => (value - extent[0]) / ((extent[1] - extent[0]) || 1);

    d3.csv(csvPath, d3.autoType).then(data => {
      // 先找出两列的最小值和最大值
      const xExtent = d3.extent(data, d => d[xKey]);
      const yExtent = d3.extent(data, d => d[yKey]);

      // 再把每一行数据归一化，得到新的 x 和 y
      data.forEach(d => {
        d.x = normalize(d[xKey], xExtent);
        d.y = normalize(d[yKey], yExtent);
      });

      // 归一化后，坐标轴统一用 0~1
      const x = d3.scaleLinear().domain([0, 1]).range([0, width]);
      const y = d3.scaleLinear().domain([0, 1]).range([height, 0]);

      // X 轴
      g.append("g")
        .attr("transform", `translate(0,${height})`)
        .call(d3.axisBottom(x));

      // Y 轴
      g.append("g").call(d3.axisLeft(y));

      // 散点
      g.selectAll("circle")
        .data(data)
        .enter()
        .append("circle")
        .attr("cx", d => x(d.x))
        .attr("cy", d => y(d.y))
        .attr("r", dotRadius)
        .attr("fill", dotColor)
        .attr("opacity", 0.8);
    });
  </script>
</body>

  <script>
    var svg = d3.select("svg");
    var width = 800;
    var height = 500;
    var padding = 60;

    d3.csv("iris.csv").then(function(data) {

      // 1. 取出高维数据：4个特征
      var X = data.map(function(d) {
        return [
          +d["Sepal.Length"],
          +d["Sepal.Width"],
          +d["Petal.Length"],
          +d["Petal.Width"]
        ];
      });

      // 2. PCA降维：4维降到2维
      var points = pca(X);

      // 3. 把降维后的二维坐标放回数据中
      data.forEach(function(d, i) {
        d.x = points[i][0];
        d.y = points[i][1];
        d.type = d["Species"];
      });

      // 4. 设置比例尺
      var xScale = d3.scaleLinear()
        .domain(d3.extent(data, function(d) { return d.x; }))
        .range([padding, width - padding]);

      var yScale = d3.scaleLinear()
        .domain(d3.extent(data, function(d) { return d.y; }))
        .range([height - padding, padding]);

      var color = d3.scaleOrdinal()
        .domain(["setosa", "versicolor", "virginica"])
        .range(["red", "green", "blue"]);

      // 5. 画散点
      svg.selectAll("circle")
        .data(data)
        .enter()
        .append("circle")
        .attr("cx", function(d) { return xScale(d.x); })
        .attr("cy", function(d) { return yScale(d.y); })
        .attr("r", 5)
        .attr("fill", function(d) { return color(d.type); });

      // 6. 添加坐标轴
      svg.append("g")
        .attr("transform", "translate(0," + (height - padding) + ")")
        .call(d3.axisBottom(xScale));

      svg.append("g")
        .attr("transform", "translate(" + padding + ",0)")
        .call(d3.axisLeft(yScale));

      // 7. 添加标题
      svg.append("text")
        .attr("x", width / 2)
        .attr("y", 30)
        .attr("text-anchor", "middle")
        .attr("font-size", 22)
        .text("202314109方昕哲 PCA降维散点图");
    });


    // PCA简化函数：高维数据降成二维
    function pca(X) {
      var n = X.length;
      var m = X[0].length;

      // 求每一列的平均值
      var mean = [];
      for (var j = 0; j < m; j++) {
        mean[j] = d3.mean(X, function(d) { return d[j]; });
      }

      // 数据中心化
      var Z = X.map(function(row) {
        return row.map(function(v, j) {
          return v - mean[j];
        });
      });

      // 简化PCA投影：使用两组主方向系数
      var pc1 = [0.5, -0.3, 0.6, 0.5];
      var pc2 = [0.4, 0.8, -0.2, -0.4];

      // 计算二维坐标
      return Z.map(function(row) {
        return [
          dot(row, pc1),
          dot(row, pc2)
        ];
      });
    }

    // 向量点乘函数
    function dot(a, b) {
      var sum = 0;
      for (var i = 0; i < a.length; i++) {
        sum += a[i] * b[i];
      }
      return sum;
    }
  </script>

import matplotlib.pyplot as plt
import numpy as np

# 中文字体
plt.rcParams["font.sans-serif"] = [
    "Arial Unicode MS",
    "Microsoft YaHei",
    "SimHei",
    "PingFang SC",
    "Heiti SC"
]
plt.rcParams["axes.unicode_minus"] = False

# 数据
types = ["哲学", "历史", "教育", "科技", "文学", "经济"]
A = [25, 20, 36, 40, 75, 90]
B = [35, 26, 45, 50, 35, 66]

# 柱状图
x = np.arange(len(types))
w = 0.35

plt.bar(x - w / 2, A, w, label="商家A")
plt.bar(x + w / 2, B, w, label="商家B")

plt.xticks(x, types)
plt.xlabel("图书类别")
plt.ylabel("销售量")
plt.title("202314109方昕哲图书销售量")
plt.legend()

plt.show()

import matplotlib.pyplot as plt
import numpy as np

# 设置中文字体，兼容 Windows 和 Mac
plt.rcParams["font.sans-serif"] = [
    "Microsoft YaHei", "SimHei", "PingFang SC", "Arial Unicode MS"
]
plt.rcParams["axes.unicode_minus"] = False

# 数据
classes = ["1班", "2班", "3班", "4班"]
boys = [5, 7, 3, 6]
girls = [3, 4, 7, 2]

# 设置柱子位置
x = np.arange(len(classes))
w = 0.35

# 画堆叠柱状图
plt.bar(x, boys, width=w, label="男生")
plt.bar(x, girls, width=w, bottom=boys, label="女生")

# 设置标题和坐标轴
plt.title("202314109方昕哲绘制男女人数对比")
plt.xlabel("班级")
plt.ylabel("人数")
plt.xticks(x, classes)

# 设置图例
plt.legend()

# 显示图形
plt.show()
